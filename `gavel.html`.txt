<script>
// --- STATE ---
let currentCategory = '';
let currentFile = null;
let currentCondition = 'Good';
let logs = [];
const LS_KEY = 'gavel_logs_v1';

const CAT_EMOJI = {
  'Furniture':'🪑', 'Electrical':'⚡', 'Bric-a-brac':'🏺', 'Clothes':'👕',
  'Books':'📚', 'Tools':'🔧', 'Toys':'🧸', 'Misc':'📦'
};

// --- INIT ---
function init() {
  try { logs = JSON.parse(localStorage.getItem(LS_KEY)) || []; } catch(e) { logs = []; }
  renderHistory();

  document.getElementById('fileInput').addEventListener('change', e => {
    if (e.target.files && e.target.files[0]) {
      currentFile = e.target.files[0];
      const url = URL.createObjectURL(currentFile);
      const preview = document.getElementById('preview');
      preview.src = url;
      preview.style.display = 'block';
    }
  });

  // Register offline service worker when hosted (not file://)
  if ('serviceWorker' in navigator && window.location.protocol !== 'file:') {
    const swCode = `self.addEventListener('install',e=>self.skipWaiting());self.addEventListener('activate',e=>e.waitUntil(clients.claim()));self.addEventListener('fetch',e=>e.respondWith(fetch(e.request).catch(()=>new Response('Offline - no wifi',{status:503,headers:{'Content-Type':'text/plain'}}))))`;
    const blob = new Blob([swCode], {type:'application/javascript'});
    navigator.serviceWorker.register(URL.createObjectURL(blob)).catch(()=>{});
  }
}

// --- NAVIGATION ---
function showView(id) {
  ['view-home','view-capture','view-history'].forEach(v => {
    document.getElementById(v).style.display = (v === id) ? 'block' : 'none';
  });
}

function selectCategory(cat) {
  currentCategory = cat;
  document.getElementById('catLabel').textContent = (CAT_EMOJI[cat]||'') + ' ' + cat;
  resetCapture(false);
  showView('view-capture');
}

function resetCapture(goHome) {
  currentFile = null;
  currentCondition = 'Good';
  document.getElementById('fileInput').value = '';
  document.getElementById('preview').style.display = 'none';
  document.getElementById('itemName').value = '';
  setCondition('Good');
  if (goHome) showView('view-home');
}

// --- CONDITION ---
function setCondition(cond) {
  currentCondition = cond;
  document.querySelectorAll('.condition-btn').forEach(b => {
    b.classList.toggle('active', b.dataset.cond === cond);
  });
}

// --- EBAY URL ---
function buildEbayUrl(query) {
  const q = encodeURIComponent(query.trim() || currentCategory);
  return `https://www.ebay.co.uk/sch/i.html?_nkw=${q}&LH_Sold=1&LH_Complete=1&_sop=13`;
}

// --- ACTIONS ---
function shareToLens() {
  if (!currentFile) { alert('Snap a photo first.'); return; }
  if (navigator.canShare && navigator.canShare({files: [currentFile]})) {
    navigator.share({ title: 'Identify with Lens', text: 'What is this thing?', files: [currentFile] }).catch(()=>{});
  } else {
    alert('Cannot auto-share from this file.\n\nTip: Open the Google app, tap the Lens icon, and photograph this item.');
  }
}

function checkEbay() {
  const name = document.getElementById('itemName').value;
  window.open(buildEbayUrl(name), '_blank');
}

function logIt() {
  const name = document.getElementById('itemName').value.trim() || currentCategory;
  const entry = {
    id: Date.now(),
    category: currentCategory,
    name: name,
    condition: currentCondition,
    ebayUrl: buildEbayUrl(name),
    time: new Date().toISOString()
  };
  logs.unshift(entry);
  if (logs.length > 500) logs.pop();
  localStorage.setItem(LS_KEY, JSON.stringify(logs));
  renderHistory();
  alert('Logged: ' + name + ' (' + currentCondition + ')');
  resetCapture(true);
}

// --- HISTORY ---
function renderHistory() {
  const container = document.getElementById('historyList');
  if (!logs.length) {
    container.innerHTML = '<div class="empty-state">No items yet.<br>Go scan some donations.</div>';
    return;
  }
  container.innerHTML = logs.map(log => {
    const date = new Date(log.time).toLocaleString('en-GB',{day:'numeric',month:'short',hour:'2-digit',minute:'2-digit'});
    const condColor = log.condition==='Good'?'#81c784':log.condition==='Fair'?'#d4a373':'#e57373';
    return `<div class="log-entry">
      <h3>${escapeHtml(log.name)}</h3>
      <div class="log-meta">${CAT_EMOJI[log.category]||''} ${escapeHtml(log.category)} • <span style="color:${condColor};font-weight:600;">${log.condition}</span> • ${date}</div>
      <div class="log-actions">
        <button onclick="openLogEbay(${log.id})">🏷️ eBay Sold</button>
        <button onclick="deleteLog(${log.id})">❌ Delete</button>
      </div>
    </div>`;
  }).join('');
}

function openLogEbay(id) {
  const log = logs.find(l => l.id === id);
  if (log) window.open(log.ebayUrl, '_blank');
}

function deleteLog(id) {
  logs = logs.filter(l => l.id !== id);
  localStorage.setItem(LS_KEY, JSON.stringify(logs));
  renderHistory();
}

function clearAll() {
  if (confirm('Wipe the whole log?')) {
    logs = [];
    localStorage.removeItem(LS_KEY);
    renderHistory();
  }
}

function escapeHtml(text) {
  const div = document.createElement('div');
  div.textContent = text;
  return div.innerHTML;
}

init();
